\begin{labbox}{Название работы}
Текст задания с \cmd{docker ps -a}.
\end{labbox}

\begin{code}[listing options={language=dockerfile}]
FROM nginx:1.27-alpine
COPY ./site /usr/share/nginx/html
EXPOSE 80
# комментарий на русском
CMD ["nginx", "-g", "daemon off;"]
\end{code}

\begin{code}[listing options={language=bash}]
$ docker run -d --name web -p 8080:80 nginx:alpine
$ docker logs -f web
\end{code}

\begin{code}[listing options={language=composeyaml}]
services:
  web:
    image: nginx:alpine
    ports:
      - "8080:80"
\end{code}
